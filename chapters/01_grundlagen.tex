\begin{frame}{Grundlagen: Zielumgebung Va.Si.Li-Lab}{\cite{10.1145/3489849.3489871, Mehler:et:al:2023:a}}
    \textbf{Multi-User VR-Plattform für Lernszenarien:}
    \begin{itemize}
        \item Entwickelt am Text Technology Lab der Goethe-Universität
        \item Basis: Unity + Ubiq-Framework für Netzwerk-Synchronisation
        \item Zielgruppe: Studierende für simulationsbasiertes Lernen
    \end{itemize}
    
    \vfill
    
    \textbf{Eignung als Zielumgebung:}
    \begin{itemize}
        \item Stabile VR-Infrastruktur mit Mehrbenutzerfähigkeit
        \item Modulare, erweiterbare Architektur
        \item \textbf{Bislang keine 3D-Rekonstruktion realer Umgebungen}
    \end{itemize}
\end{frame}
