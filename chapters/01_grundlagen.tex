\begin{frame}{Grundlagen: Virtual Reality}{\cite{bozkir2025eyetrackedvirtualrealitycomprehensive, MetaPassthrough2021}}
    \textbf{Was ist Virtual Reality?}
    \begin{itemize}
        \item Computergenerierte 3D-Umgebung mit vollständiger Immersion
        \item Head-Mounted Displays (HMDs): Stereoskopische Darstellung + präzises Tracking
        \item Echtzeit-Interaktion durch Kopf-, Hand- und Körpererfassung
    \end{itemize}
    
    \vspace{0.8em}
    \textbf{Moderne VR-Hardware (z.B. Meta Quest 3):}
    \begin{itemize}
        \item Inside-Out-Tracking: Kameras erfassen Umgebung $\rightarrow$ Position/Orientierung
        \item Passthrough-Funktionalität: RGB-Kameras ermöglichen Mixed Reality
        \item \textbf{Wichtig:} Keine Tiefensensoren $\Rightarrow$ monokulare Rekonstruktion nötig
    \end{itemize}
\end{frame}

\begin{frame}{Grundlagen: 3D-Rekonstruktion}{\cite{HERRERAGRANDA2024e37356, sun2021neuralreconrealtimecoherent3d, gao2023visfusionvisibilityawareonline3d, murai2025mast3rslamrealtimedenseslam, liu2025slam3rrealtimedensescene}}
    \textbf{Der Prozess:}
    \begin{itemize}
        \item 2D-Bilder $\rightarrow$ 3D-Geometrie
        \item Tiefe aus Kamerabewegung + Bildsequenzen (monokulare Rekonstruktion)
    \end{itemize}
    
    \vspace{0.8em}
    \textbf{Zentrale Unterscheidung nach Pose-Anforderung:}
    \begin{itemize}
        \item \textbf{Externe Posen} (z.\,B. NeuralRecon, VisFusion): System liefert Kameraposen
        \item \textbf{Interne Pose-Schätzung} (z.\,B. MASt3R-SLAM, SLAM3R): Modell schätzt Posen selbst
    \end{itemize}
    
    \vspace{0.8em}
    \textbf{Implikation für RTReconstruct:}
    \begin{itemize}
        \item Diese Heterogenität erfordert flexible System-Architektur
        \item Beide Kategorien müssen über einheitliche Schnittstelle integrierbar sein
    \end{itemize}
\end{frame}

    \begin{itemize}
        \item Basis-Technologie: Unity + Ubiq-Framework
        \item Designed für kollaborative, simulationsbasierte Lernszenarien
        \item Bereits vorhanden: Tracking, Netzwerk-Sync, Szenenlogik

    \begin{itemize}
