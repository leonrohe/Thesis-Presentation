\begin{frame}{Motivation: 3D-Rekonstruktion in VR}{(\cite{enhancinginteractions,Stotko_2019,khenak2020spatial})}
    \textbf{Ausgangssituation:}
    \begin{itemize}
        \item VR-Anwendungen werden immer immersiver
        \item Dennoch basieren sie oft auf künstlich erstellten Szenen
        \item Viele Anwendungsfälle profitieren von Bezug zur realen Welt
    \end{itemize}
    
    \vfill
    
    \textbf{Vorteile durch Rekonstruktion realer Umgebungen:}
    \begin{itemize}
        \item Spontane Erfassung und gemeinsame Analyse von Räumen
        \item Kollaborative Aufgaben ohne manuelle Modellierung
        \item Grundbaustein für dynamische, interaktive VR-Systeme
    \end{itemize}

    \vfill

    \begin{tcolorbox}[box5]
        \centering
        \textbf{Idee:}~Rekonstruktion mithilfe von Photogrammetrie
    \end{tcolorbox}
\end{frame}

\begin{frame}{Bestehende Ansätze zur 3D-Rekonstruktion}
    \textbf{Klassische Photogrammetrie (Offline/Batch):}
    \begin{itemize}
        \item Hochauflösende Ergebnisse
        \item Umfangreiche Bildaufnahmen, lange Verarbeitungszeiten
        \item Nicht geeignet für interaktive VR-Szenarien
    \end{itemize}
    
    \vfill
    
    \textbf{RGB-D-basierte Online-Rekonstruktion:}
    \begin{itemize}
        \item Echtzeitfähig durch Tiefensensoren
        \item Hardware-Abhängigkeit (bei Consumer-VR oft nicht verfügbar)
    \end{itemize}
    
    \vfill
    
    \textbf{$\Longrightarrow$ Monokulare Echtzeitverfahren}
\end{frame}

\begin{frame}{Monokulare Echtzeit-Rekonstruktion: Potenzial und Herausforderungen}{(\cite{sun2021neuralreconrealtimecoherent3d,gao2023visfusionvisibilityawareonline3d,liu2025slam3rrealtimedensescene, murai2025mast3rslamrealtimedenseslam})}
    \textbf{Potenzial:}
    \begin{itemize}
        \item Keine zusätzliche Hardware erforderlich (nur RGB-Kamera)
        \item Direkte Nutzung von VR-Headset-Frontkameras
        \item Verschiedene Ansätze verfügbar (SLAM, volumetrisch, lernbasiert)
    \end{itemize}
    
    \vfill
    
    \textbf{Herausforderungen für VR-Integration:}
    \begin{itemize}
        \item Heterogene Laufzeitumgebungen und Frameworks
        \item Fehlende Streaming-Infrastruktur für Multi-User-VR
        \item Unterschiedliche Schnittstellen und Pose-Anforderungen
    \end{itemize}
\end{frame}
