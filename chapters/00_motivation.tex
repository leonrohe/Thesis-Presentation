\section{Motivation}\sectionFrame

\begin{frame}{3D-Rekonstruktion in VR}{(\cite{enhancinginteractions,Stotko_2019,khenak2020spatial})}
    \textbf{Ausgangssituation:}
    \begin{itemize}
        \item VR-Anwendungen werden immer immersiver
        \item Dennoch basieren sie oft auf künstlich erstellten Szenen
        \item Viele Anwendungsfälle profitieren von Bezug zur realen Welt
    \end{itemize}
    
    \bigskip
    
    \textbf{Vorteile durch Rekonstruktion realer Umgebungen:}
    \begin{itemize}
        \item Spontane Erfassung und gemeinsame Analyse von Räumen
        \item Kollaborative Aufgaben ohne manuelle Modellierung
        \item Grundbaustein für dynamische, interaktive VR-Systeme
    \end{itemize}
\end{frame}

\begin{frame}{Bestehende Ansätze zur 3D-Rekonstruktion}
    \textbf{Klassische Photogrammetrie (Offline/Batch):}
    \begin{itemize}
        \item Hochauflösende Ergebnisse
        \item Umfangreiche Bildaufnahmen, lange Verarbeitungszeiten
        \item Nicht geeignet für interaktive VR-Szenarien
    \end{itemize}
    
    \vfill
    
    \textbf{RGB-D-basierte Online-Rekonstruktion:}
    \begin{itemize}
        \item Echtzeitfähig durch Tiefensensoren
        \item Hardware-Abhängigkeit (bei Consumer-VR oft nicht verfügbar)
    \end{itemize}
    
    \vfill
    
    \begin{tcolorbox}[box5]
        \centering
        \textbf{Lösung:}~Monokulare Echtzeitverfahren
    \end{tcolorbox}
\end{frame}

\begin{frame}{Monokulare Echtzeit-Rekonstruktion: Potenzial}{(\cite{sun2021neuralreconrealtimecoherent3d,gao2023visfusionvisibilityawareonline3d,liu2025slam3rrealtimedensescene, murai2025mast3rslamrealtimedenseslam})}
    \textbf{Vorteile:}
    \begin{itemize}
        \item Keine zusätzliche Hardware erforderlich (nur RGB-Kamera)
        \item Direkte Nutzung von VR-Headset-Frontkameras
        \item Inkrementelle Rekonstruktion während der Bewegung
        \item Verschiedene Ansätze verfügbar (SLAM, volumetrisch, lernbasiert)
    \end{itemize}
    
    \bigskip
    
    \textbf{Aber: Praktische Integration in VR kaum etabliert}
    \begin{itemize}
        \item Modelle in isolierten Forschungskontexten entwickelt
        \item Heterogene Laufzeitumgebungen und Frameworks
        \item Nicht auf interaktive Multi-User-VR ausgelegt
    \end{itemize}
\end{frame}

\begin{frame}{Herausforderungen der Integration}
    \textbf{Was die Integration erschwert:}
    \begin{itemize}
        \item Eigene Laufzeitumgebungen, Frameworks und Pipelines
        \item Fehlende Streaming-Infrastruktur für kontinuierliche Updates
        \item Unterschiedliche Schnittstellen und Ausgabeformate
        \item Pose-Anforderungen variieren (intern vs.\ extern)
    \end{itemize}
    
    \bigskip
    
    \textbf{Folge:}
    \begin{itemize}
        \item Hoher manueller Integrationsaufwand pro Verfahren
        \item Kein systematischer Vergleich unter identischen Bedingungen
    \end{itemize}
\end{frame}

\begin{frame}{Forschungslücke}
    \textbf{Bisherige Forschung fokussiert auf:}
    \begin{itemize}
        \item Verbesserung einzelner Rekonstruktionsalgorithmen
        \item Qualität und Geschwindigkeit spezifischer Verfahren
    \end{itemize}
    
    \bigskip
    
    \textbf{Unzureichend adressiert:}
    \begin{itemize}
        \item Wiederverwendbare Infrastruktur für VR-Integration
        \item Systematischer Vergleich heterogener Verfahren
        \item Deployment-Architekturen für praktischen Einsatz
    \end{itemize}
\end{frame}

\begin{frame}{Forschungsfrage}
    \vfill
    \begin{center}
        \large
        \textit{Wie gut eignet sich eine modulare, containerisierte Systemarchitektur zur\\
        Integration verschiedener Echtzeit-3D-Rekonstruktionsverfahren\\
        in eine bestehende Virtual-Reality-Umgebung?}
    \end{center}
    \vfill
\end{frame}
