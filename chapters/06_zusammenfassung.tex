\section{Diskussion \& Fazit}\sectionFrame

\begin{frame}{Echtzeitfähigkeit und GPU-Engpässe}{}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Volumetrische Verfahren}
        \begin{itemize}
            \itemsep0.4em
            \item Durchsatz: 0,30-0,62 fps
            \item GPU-Auslastung: 43-56\%
            \item Detaillierte Oberflächenqualität
            \item Geschlossene Meshes
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{SLAM-basierte Verfahren}
        \begin{itemize}
            \itemsep0.4em
            \item Durchsatz: 0,03-0,13 fps
            \item GPU-Auslastung: 98-100\%
            \item VRAM-Bottleneck: 7,9-8,0 GB
            \item GTX 1070 Ti limitiert
        \end{itemize}
    \end{columns}
    
    \vfill
    \textbf{Architektur-Overhead minimal}
    \begin{itemize}
        \itemsep0.4em
        \item Inferenz dominiert Gesamtlatenz: 63-90\%
        \item Netzwerk + Rendering konstant <1s
    \end{itemize}
\end{frame}

\begin{frame}{Modularität: Containerisierung ermöglicht Integration}{}
    \textbf{Erfolgreiche Integration heterogener Verfahren}
    \begin{itemize}
        \itemsep0.4em
        \item 4 Modelle mit konfliktären PyTorch-Versionen isoliert betrieben
        \item Integrationsaufwand: 4-8 Stunden pro Modell
        \item BaseReconstructionModel kapselt WebSocket-Logik vollständig
        \item Docker-Container: 47-66 Zeilen, Worker: 205-401 Zeilen
    \end{itemize}
    
    \vspace{0.8em}
    
    \textbf{Systemstabilität unter Produktionsbedingungen}
    \begin{itemize}
        \itemsep0.4em
        \item Router-Overhead minimal: 20\% CPU, 100 MB RAM
        \item Multi-User-Stabilität: 2 Szenen, 4 Clients parallel über 6h
        \item 0 Verbindungsabbrüche in 60 Testläufen
    \end{itemize}
\end{frame}


\begin{frame}{Rekonstruktionsqualität und SDK-Limitationen}{}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Synthetische Szenen}
        \begin{itemize}
            \itemsep0.4em
            \item Volumetrisch stabil (F: 0,41-0,68)
            \item VisFusion konsistent (F: 0,54-0,62)
            \item SLAM: Drift-Artefakte (F: 0,42-0,63)
            \item SLAM3R: V1$\longrightarrow$V3 Rückgang 31\%
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Reale VR-Szenen}
        \begin{itemize}
            \itemsep0.4em
            \item \textbf{Leistungsumkehr}: SLAM überlegen
            \item Volumetrisch fragmentiert
            \item MASt3R-SLAM trotz 0,03-0,10 fps beste Vollständigkeit
        \end{itemize}
    \end{columns}
    
    \vfill
    \begin{tcolorbox}[box5, title={\textbf{Kernursachen}}]
        Meta Quest 3 SDK liefert keine Frame-Timestamps
        \begin{itemize}
            \item zeitliche Inkonsistenzen
            \item TSDF-Artefakte
        \end{itemize}
        SLAM kompensiert durch Pose-Optimierung.
        \begin{itemize}
            \item SDK-Limitation, nicht Algorithmen-Schwäche.
        \end{itemize}
    \end{tcolorbox}
\end{frame}

\begin{frame}{Beantwortung der Forschungsfrage}{}
    \begin{center}
        \textit{Wie gut eignet sich eine modulare, containerisierte Architektur zur\\Integration heterogener Rekonstruktionsverfahren in VR?}
    \end{center}
    
    \vfill
    
    \textbf{Antwort: Sehr gut für Forschungs- und Evaluationskontexte}
    \begin{itemize}
        \itemsep0.4em
        \item Standardisierte Schnittstellen abstrahieren Framework-Heterogenität erfolgreich
        \item Asynchrone Streaming-Architektur reduziert Kopplungsgrad signifikant
        \item Flexible Hardware-Nutzung ohne Code-Anpassungen (Docker + GPU-Isolation)
        \item Kontextspezifische Verfahrenswahl ermöglicht (volumetrisch vs. SLAM)
        \item Va.Si.Li-Lab: Additive Integration ohne Mehrbenutzer-Eingriff
    \end{itemize}
    
    \vfill
    \begin{tcolorbox}[box5, boxsep=0.5em]
        \small
        \textbf{Wissenschaftlich:} Systematische Vergleiche unter identischen Bedingungen\\
        \textbf{Praktisch:} Wiederverwendbare Infrastruktur (4-8h Integrationsaufwand/Modell)
    \end{tcolorbox}
\end{frame}

\begin{frame}{Wissenschaftlicher Beitrag}{}
    \textbf{Methodisch}
    \begin{itemize}
        \itemsep0.4em
        \item Erste systematische Vergleichsstudie monokularer Rekonstruktion in VR-Headsets
        \item Identifikation SDK-bedingter Limitationen (Quest 3 Timestamps)
        \item Nachweis: Hardware-Einschränkungen dominieren über Algorithmen
    \end{itemize}

    \vfill

    \textbf{Praktisch}
    \begin{itemize}
        \itemsep0.4em
        \item Wiederverwendbare Infrastruktur (4-8h Integrationsaufwand/Modell)
        \item Deployment-Referenzarchitektur für VR-Forschungsprototypen
    \end{itemize}
\end{frame}

\begin{frame}{Limitationen}{}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Performance und Hardware}
        \begin{itemize}
            \itemsep0.4em
            \item Durchsatz 0,03-0,62 fps unzureichend für Echtzeit-VR
            \item GPU-Engpass: 8 GB VRAM limitiert Parallelbetrieb
            \item SLAM: 98-100\% GPU-Auslastung
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{System und SDK}
        \begin{itemize}
            \itemsep0.4em
            \item Keine Persistenz: Container-Neustarts verlieren Rekonstruktionszustand
            \item Manuelle SLAM-Skalierung erforderlich
            \item Meta Quest 3: Fehlende Timestamps
        \end{itemize}
    \end{columns}
    
    \vfill
    \begin{tcolorbox}[box5, boxsep=0.5em]
        \textbf{Kern-Trade-off:} Geschwindigkeit vs. Robustheit - Kontextabhängige Verfahrenswahl notwendig
    \end{tcolorbox}
\end{frame}
