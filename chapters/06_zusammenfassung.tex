\begin{frame}{Diskussion: Echtzeitfähigkeit und Engpässe}
    \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l|>{\centering}p{3.5cm}|>{\centering\arraybackslash}p{3.5cm}}
        \textbf{Metrik} & \textbf{Volumetrisch} & \textbf{SLAM-basiert} \\
        \midrule
        Durchsatz & 0,30--0,62 FPS & 0,03--0,13 FPS \\
        GPU-Auslastung & 43--56\% & 98--100\% \\
        VRAM-Nutzung & 3,8--8,0 GB & 8,0 GB (Limit) \\
    \end{tabular}
    \end{center}
    
    \vfill
    
    \textbf{Bottleneck}:
    \begin{itemize}
        \item Hardware (GTX 1070 Ti) limitiert SLAM-Verfahren
    \end{itemize}    
    \textbf{Architektur}:
    \begin{itemize}
        \item Inferenz dominiert Latenz (63--90\%), Netzwerk + Rendering <1s
    \end{itemize}
\end{frame}

\begin{frame}{Diskussion: Modularität}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Integration}
        \begin{itemize}
            \item Integration inkompatibler Modelle
            \item Standardisierte Schnittstellen
            \item 2-8h Aufwand pro Modell
        \end{itemize}
        
        \column{0.48\textwidth}
        \textbf{Stabilität}
        \begin{itemize}
            \item Multi-User-Betrieb problemlos
            \item 0/60 Verbindungsabbrüche, 6h Betrieb
            \item geringer Overhead (25\% CPU, 100 MB RAM)
        \end{itemize}
    \end{columns}
    
    \vfill
    
    \begin{tcolorbox}[box5, boxsep=0.5em]
        \centering
        Modulare Architektur ermöglicht produktiven Forschungsbetrieb
    \end{tcolorbox}
\end{frame}

\begin{frame}{Diskussion: Rekonstruktionsqualität und SDK-Limitationen}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Synthetische Szenen}
        \begin{itemize}
            \item Volumetrisch stabil\\
            {\small \textit{(F-Score: 0,41--0,68)}}
            \item VisFusion am konsistentesten\\
            {\small \textit{(F-Score: 0,54--0,62)}}
            \item SLAM mit Drift-Artefakten
        \end{itemize}
        
        \column{0.48\textwidth}
        \textbf{Reale VR-Szenen}
        \begin{itemize}
            \item \textbf{Leistungsumkehr}: SLAM überlegen
            \item Volumetrisch fragmentiert
            \item MASt3R-SLAM trotz niedriger FPS beste Vollständigkeit
        \end{itemize}
    \end{columns}
    
    \vfill
    
    \begin{tcolorbox}[box5, boxsep=0.5em, title={\textbf{Kernursache: SDK-Limitation}}]
        Meta Quest 3 SDK lieferte keine Frame-Timestamps
        \begin{itemize}
            \item Zeitliche Inkonsistenzen führten zu TSDF-Artefakten
            \item SLAM kompensierte durch Pose-Optimierung
        \end{itemize} 
    \end{tcolorbox}
\end{frame}

\begin{frame}{Fazit: Beantwortung der Forschungsfrage}{}
    \vfill
    \begin{center}
        \large
        \textit{Wie gut eignet sich eine modulare, containerisierte Systemarchitektur zur\\
        Integration verschiedener Echtzeit-3D-Rekonstruktionsverfahren\\
        in eine bestehende Virtual-Reality-Umgebung?}
    \end{center}
    \vspace{2cm}
    \begin{tcolorbox}[box5]
        \centering
        \textbf{Antwort:}~Hohe Eignung - insbesondere für Forschungs- und Evaluationskontexte
    \end{tcolorbox} 
\end{frame}

\begin{frame}{Fazit: Beantwortung der Forschungsfrage}{}
    \textbf{Begründung:}
    \begin{enumerate}
        \item Standardisierte Schnittstellen ermöglichen Framework-unabhängige Integration
        \item Asynchrone Architektur minimiert Abhängigkeiten zwischen Komponenten
        \item Container-Isolation erlaubt parallele GPU-Nutzung unterschiedlicher Verfahren
        \item Modularer Aufbau ermöglicht flexible Verfahrenswahl je Anwendungsfall
    \end{enumerate}
    
    \vfill
    
    \textbf{Validierung:}
    \begin{itemize}
        \item Erfolgreiche Integration von 4 heterogenen Verfahren
        \item Deployment in Va.Si.Li-Lab ohne Änderungen an bestehender Mehrbenutzerlogik
        \item Integrationsaufwand: 2--8 Stunden pro neuem Rekonstruktionsverfahren
        \item Systematische Vergleichbarkeit unter identischen Bedingungen nachgewiesen
    \end{itemize}
\end{frame}

\begin{frame}{Fazit: Wissenschaftlicher Beitrag}{}
    \textbf{Methodisch:}
    \begin{itemize}
        \itemsep0.4em
        \item Systematischer Vergleich heterogener Verfahren unter identischen Bedingungen
        \item Identifikation SDK-bedingter Limitationen (Quest 3: fehlerhafte Timestamps)
        \item Trade-off-Analyse: volumetrisch (schnell, hardware-sensitiv) vs. SLAM (langsam, robust)
    \end{itemize}
    
    \vfill
    
    \textbf{Praktisch:}
    \begin{itemize}
        \itemsep0.4em
        \item Wiederverwendbare Infrastruktur für VR-Rekonstruktionsforschung
        \item Referenzarchitektur für containerisierte Forschungsprototypen
        \item Open-Source-Bereitstellung zur Nachnutzung
    \end{itemize}
\end{frame}

\begin{frame}{Fazit: Limitationen}
    \textbf{1. Performance-Limitationen}
    \begin{itemize}
        \itemsep0.4em
        \item Durchsatz: 0,03-0,62 FPS $\rightarrow$ nicht \lq wirklich\rq ~echtzeittauglich
        \item SLAM-Verfahren: 98-100\% GPU-Auslastung
        \item VRAM-Bottleneck: 8 GB limitieren Parallelbetrieb
    \end{itemize}
    
    \vfill
    
    \textbf{2. Architektur-Limitationen}
    \begin{itemize}
        \itemsep0.4em
        \item Keine Persistenz bei Container-Neustarts
        \item Rekonstruktionen werden nicht inkrementell gesendet
        \item SLAM-Rekonstruktionen benötigen manuelle Registrierung
    \end{itemize}
    
    \vfill
    
    \textbf{3. Hardware-/SDK-Limitationen}
    \begin{itemize}
        \itemsep0.3em
        \item Meta Quest 3: Fehlende Frame-Timestamps erschweren Synchronisation
    \end{itemize}
\end{frame}
